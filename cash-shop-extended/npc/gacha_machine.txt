// Cash Gacha Machine
// By: Lil Art
// ============================================================================
// Exchange Zeny, Cash Points, or Cachua Coupons for one random item from the
// Cash Shop catalog (db/item_cash.yml).
//
// How to customize:
// - Costs: .zeny_cost, .cash_cost, .coupon_cost in OnInit.
// - The pool: one "AegisName:weight" line per item in OnInit, the same
//   AegisNames as db/item_cash.yml. A higher weight is a more common item.
//   The default weight is 15000 / the item's shop price, so cheap items
//   come up often and expensive ones rarely.
// - An item the server does not know is skipped at start-up and named in
//   the map-server log ("[Gacha] unknown item ...").
// ============================================================================

prontera,172,177,4	script	Gacha Machine#gachamach	2_SLOT_MACHINE,{
	mes "[Gacha Machine]";
	mes "Welcome! Would you like to try your luck today?";
	mes "Cost per pull:";
	mes "^0055FF- " + callfunc("F_InsertComma", .zeny_cost) + " Zeny^000000";
	mes "^0055FF- " + .cash_cost + " Cash Points^000000";
	mes "^0055FF- " + .coupon_cost + " Cachua Coupons^000000";
	next;

	.@pay = select("Pay with Zeny:Pay with Cash Points:Pay with Kachua Coupons:Cancel");
	if (.@pay == 4)
		close;

	if (.item_count <= 0 || .total_weight <= 0) {
		mes "[Gacha Machine]";
		mes "The item pool is currently empty. Please contact an Administrator.";
		close;
	}

	// Roll first, so nothing is paid for an item the player cannot carry.
	.@roll = rand(1, .total_weight);
	.@won_item = .item_id[.item_count - 1];
	for (.@i = 0; .@i < .item_count; .@i++) {
		.@roll -= .item_weight[.@i];
		if (.@roll <= 0) {
			.@won_item = .item_id[.@i];
			break;
		}
	}

	if (!checkweight(.@won_item, 1)) {
		mes "[Gacha Machine]";
		mes "You are carrying too much. Make some room first!";
		close;
	}

	if (.@pay == 1) {
		if (Zeny < .zeny_cost) {
			mes "[Gacha Machine]";
			mes "You don't have enough Zeny!";
			close;
		}
		Zeny -= .zeny_cost;
	} else if (.@pay == 2) {
		if (#CASHPOINTS < .cash_cost) {
			mes "[Gacha Machine]";
			mes "You don't have enough Cash Points!";
			close;
		}
		#CASHPOINTS -= .cash_cost;
	} else {
		if (countitem(.coupon_id) < .coupon_cost) {
			mes "[Gacha Machine]";
			mes "You don't have enough Kachua Coupons!";
			close;
		}
		delitem .coupon_id, .coupon_cost;
	}

	specialeffect2 154;
	getitem .@won_item, 1;

	mes "[Gacha Machine]";
	mes "Congratulations! You obtained:";
	mes "^008800" + getitemname(.@won_item) + "^000000";
	close;
